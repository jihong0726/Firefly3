const vaultAmountInput = document.getElementById("vaultAmountInput"); const vaultInputLabel = document.getElementById("vaultInputLabel"); const vaultLiveHint = document.getElementById("vaultLiveHint"); // 允许输入：数字 + 一个小数点 function sanitizeMoneyInput(s){ s = String(s ?? "").replace(/[^\d.]/g, ""); const parts = s.split("."); if (parts.length > 2) s = parts[0] + "." + parts.slice(1).join(""); // 只保留第一个点 // 去掉前导多个 0，但保留 "0." 的情况 s = s.replace(/^0+(?=\d)/, "0"); return s; } function toMoneyNumber(s){ const n = Number(s); return Number.isFinite(n) ? n : NaN; } function updateVaultLiveHint(){ const raw = sanitizeMoneyInput(vaultAmountInput.value); if (raw !== vaultAmountInput.value) vaultAmountInput.value = raw; const num = toMoneyNumber(raw); if (!raw || !Number.isFinite(num) || num <= 0){ vaultLiveHint.textContent = ""; return; } if (vaultMode === "exchange_in"){ const rm = num / Number(currentRate || 1); vaultLiveHint.textContent = `将按当前汇率 1:${Number(currentRate).toFixed(3)} 折合 RM ${rm.toFixed(2)}`; } else { vaultLiveHint.textContent = `将记录为 RM ${num.toFixed(2)} 的开销`; } } // 打开弹窗时初始化输入框 function resetVaultInput(){ vaultAmountInput.value = ""; vaultLiveHint.textContent = ""; // 自动聚焦，方便直接打数字 setTimeout(()=> vaultAmountInput.focus(), 80); } function openVaultModal(mode){ vaultMode = mode; vaultEmo = (mode==="exchange_in") ? "🔄" : "🍔"; if (mode==="exchange_in"){ vaultModalIcon.textContent = "🔄"; vaultModalTitle.textContent = "换汇存入"; vaultModalSub.textContent = "输入数字即可"; vaultModeChip.textContent = "🔄 换汇存入"; vaultModeHint.textContent = "输入 TWD"; vaultInputUnit.textContent = "TWD"; vaultInputLabel.textContent = "存入金额"; emojiPick.style.display = "none"; }else{ vaultModalIcon.textContent = "🧾"; vaultModalTitle.textContent = "记录开销"; vaultModalSub.textContent = "用表情选择分类，然后输入 RM 数字"; vaultModeChip.textContent = "🧾 开销"; vaultModeHint.textContent = "输入 RM"; vaultInputUnit.textContent = "RM"; vaultInputLabel.textContent = "开销金额"; emojiPick.style.display = "flex"; } resetVaultInput(); vaultOverlay.classList.add("show"); } // 输入时实时提示折算/记录内容 vaultAmountInput.addEventListener("input", updateVaultLiveHint); // 确认：直接读取输入框 vaultConfirmBtn.addEventListener("click", ()=>{ const raw = sanitizeMoneyInput(vaultAmountInput.value); const num = toMoneyNumber(raw); if (!raw || !Number.isFinite(num) || num <= 0){ alert("请输入大于 0 的数字"); vaultAmountInput.focus(); return; } const rec = { id: `${Date.now()}_${vaultMode}_${num}`, ts: Date.now(), kind: vaultMode, amount: (vaultMode === "expense") ? Number(num.toFixed(2)) : Number(num), // 开销保留两位 emo: vaultEmo, note: "", rate: Number(currentRate) }; addVault(rec); renderVault(); closeVaultModal(); });
